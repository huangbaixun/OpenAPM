curl -L "https://lumen.dev/download/otel-agent.jar" -o otel-agent.jar

# 2. 启动应用时挂载 Agent (零代码改造)
java \
  -javaagent:./otel-agent.jar \
  -Dotel.service.name=payment-gateway \
  -Dotel.resource.attributes=deployment.environment=production \
  -Dotel.exporter.otlp.endpoint=https://otlp.lumen.dev:4317 \
  -Dotel.exporter.otlp.headers="authorization=Bearer lm_xxxx_9a12bc" \
  -Dotel.traces.sampler=parentbased_traceidratio \
  -Dotel.traces.sampler.arg=0.1 \
  -jar your-application.jar
